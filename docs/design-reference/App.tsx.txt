type View = "home" | "intro" | "game" | "ending"
type Modal = null | "journal" | "history" | "settings" | "journeys" | "collection" | "about" | "pause"
type Saved = {
  scene: number
  score: number
  answers: number[]
  k6?: boolean
  k12?: boolean
}

const emptyProgress: Saved = { scene: 0, score: 0, answers: [] }

export default function App() {
  const [view, setView] = useState<View>("home")
  const [modal, setModal] = useState<Modal>(null)
  const [progress, setProgress] = useState<Saved>(() => {
    try {
      return (
        JSON.parse(localStorage.getItem("hanh-trinh-progress") || "null") ||
        emptyProgress
      )
    } catch {
      return emptyProgress
    }
  })
  const [selected, setSelected] = useState<number | null>(null)
  const [sound, setSound] = useState(true)
  const [largeText, setLargeText] = useState(false)

  useEffect(() => {
    localStorage.setItem("hanh-trinh-progress", JSON.stringify(progress))
  }, [progress])

  const scene = scenes[progress.scene]
  const hasProgress = progress.answers.length > 0
  const ending = useMemo(() => {
    if (progress.score >= 18 && progress.k6 && progress.k12) return "good"
    if (progress.score >= 8) return "neutral"
    return "bad"
  }, [progress])

  function start(fresh = false) {
    if (fresh) setProgress(emptyProgress)
    setSelected(null)
    setView("intro")
    setModal(null)
  }

  function choose(index: number) {
    if (selected !== null) return
    setSelected(index)
    const choice = scene.choices[index]
    setProgress((p) => ({
      ...p,
      score: p.score + choice.points,
      answers: [...p.answers.slice(0, p.scene), index],
      ...(p.scene === 5 ? { k6: !!choice.correct } : {}),
      ...(p.scene === 11 ? { k12: !!choice.correct } : {}),
    }))
  }

  function next() {
    if (progress.scene === 11) {
      setView("ending")
      return
    }
    setProgress((p) => ({ ...p, scene: p.scene + 1 }))
    setSelected(null)
    window.scrollTo({ top: 0, behavior: "smooth" })
  }

  function resume() {
    setSelected(progress.answers[progress.scene] ?? null)
    setView("game")
  }

  return (
    <main className={largeText ? "large-text" : ""}>
      {view === "home" && (
        <div className="home-page">
          <header className="site-header">
            <button
              className="brand"
              onClick={() => setView("home")}
              aria-label="Trang chủ"
            >
              <span className="brand-seal">
                <Icon name="compass" size={24} />

                <b>HÀNH TRÌNH</b>

                className="icon-button"

              >
                <Icon name="gear" />
              </button>

            <div className="hero-art-wrap">
              <HarborScene />

                <span>MINH HỌA</span> Bến cảng Sài Gòn, 1911
              </div>
              <button
                className="hotspot h1"
                onClick={() => setModal("history")}
                aria-label="Khám phá con tàu"
              >
                <span>+</span>
              </button>
              <button
                className="hotspot h2"
                onClick={() => setModal("history")}
                aria-label="Khám phá chiếc túi của An"
              >
                <span>+</span>
              </button>
            </div>
            <div className="hero-content">
              <div className="eyebrow">
                <i /> TRÒ CHƠI KỂ CHUYỆN LỊCH SỬ <i />
              </div>
              <h1>
                Hành trình
                <br />
                <em>tìm đường cứu nước</em>
              </h1>
              <p className="hero-lead">
                Theo bước An — một thanh niên trẻ đi qua những bến cảng, trang
                sách và câu hỏi lớn của thời đại.

        </div>
      )}

      {view === "ending" && (
        <section className={`ending-page ${ending}`}>
          <div className="ending-art">
            <HarborScene night={ending !== "good"} />
          </div>
          <div className="ending-content paper-card">
            <span className="chapter-no">KẾT THÚC HÀNH TRÌNH I</span>
            <h2>
              {ending === "good"
                ? "Ánh sáng trên con đường"
                : ending === "neutral"
                  ? "Ngọn đèn còn sáng"
                  : "Cuốn sổ khép lại"}
            </h2>
            <div className="ending-seal">
              <Icon name={ending === "good" ? "spark" : "book"} size={30} />
            </div>
            <p>
              {ending === "good"
                ? "An chưa thể biết hết những thử thách phía trước. Nhưng mong muốn giúp quê hương đã có thêm hiểu biết, phương hướng và những người cùng bước."
                : ending === "neutral"
                  ? "Chuyến đi đã thay đổi An, nhưng chưa phải câu hỏi nào anh cũng hiểu đến nơi. Buổi học ngày mai vẫn còn một chỗ dành cho anh."
                  : "An đã đi qua nhiều nơi, nhưng các câu hỏi chưa được theo đuổi đến cùng. Ngoài kia, dòng lịch sử vẫn tiếp tục."}
            </p>
            <div className="ending-stats">
              <span>
                <b>12</b> màn đã đi qua
              </span>
              <span>
                <b>
                  {
                    progress.answers.filter((_, i) =>
                      [4, 5, 6, 9, 11].includes(i),
                    ).length
                  }
                </b>{" "}
                thẻ kiến thức
              </span>
              <span>
                <b>{progress.answers.length}</b> lựa chọn đã lưu
              </span>
            </div>
            <button
              className="primary-button"
              onClick={() => setModal("journeys")}
            >
              Tiếp tục hành trình <Icon name="arrow" />
            </button>
            <button className="text-button" onClick={() => setView("home")}>
              Trở về trang chủ
            </button>
          </div>
        </section>
      )}

      {modal && (
        <ModalView
          modal={modal}
          close={() => setModal(null)}
          progress={progress}
          setView={setView}
          start={start}
          sound={sound}
          setSound={setSound}
          largeText={largeText}
          setLargeText={setLargeText}
        />
      )}
    </main>
  )
}

        {modal === "about" && (
          <>
            <p>
              “Hành trình tìm đường cứu nước” là trò chơi kể chuyện giáo dục

              <span>5 tháng 6, 1911</span>
              <span>Bến cảng Sài Gòn</span>
              <span>TƯ LIỆU LỊCH SỬ</span>
            </div>
            <p>
              Ngày 5/6/1911, Nguyễn Tất Thành rời bến cảng Sài Gòn trên tàu
              Amiral Latouche-Tréville, bắt đầu hành trình tìm đường cứu nước.
            </p>
            <a
              href="https://tphcm.chinhphu.vn/hieu-them-ve-lua-chon-lich-su-cua-lanh-tu-nguyen-ai-quoc-ho-chi-minh-10121374.htm"
              target="_blank"
              rel="noreferrer"

                thế vai trò của nhân vật lịch sử.
              </span>

              <button>Vật phẩm</button>
